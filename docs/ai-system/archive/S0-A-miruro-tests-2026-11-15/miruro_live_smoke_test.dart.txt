import 'dart:io';

import 'package:anime_app/data/player/providers/miruro/miruro_catalog.dart';
import 'package:anime_app/data/player/providers/miruro/miruro_catalog_api.dart';
import 'package:anime_app/domain/player/providers/playback_provider_contract.dart';
import 'package:dio/dio.dart';
import 'package:flutter_test/flutter_test.dart';

void main() {
  test(
    'probes the public Miruro config endpoint and prints a sanitized outcome',
    () async {
      final Dio dio = Dio(
        BaseOptions(
          connectTimeout: const Duration(seconds: 8),
          receiveTimeout: const Duration(seconds: 12),
          followRedirects: false,
          maxRedirects: 0,
        ),
      );
      try {
        final Map<String, Object?> response = await DioMiruroCatalogApi(dio)
            .fetchConfig(PlaybackCancellationToken());
        final List<MiruroProviderDefinition> providers =
            MiruroCatalogParser.parseConfig(response);
        stdout.writeln(
          'MIRURO_LIVE outcome=success provider_count=${providers.length}',
        );
        expect(providers, isNotEmpty);
      } on MiruroCatalogApiException catch (error) {
        stdout.writeln(
          'MIRURO_LIVE outcome=blocked_or_unavailable status=${error.statusCode ?? 'transport'} kind=${error.kind.name}',
        );
        expect(
          error.kind,
          anyOf(
            PlaybackProviderFailureKind.unauthorized,
            PlaybackProviderFailureKind.unavailable,
            PlaybackProviderFailureKind.network,
            PlaybackProviderFailureKind.timeout,
            PlaybackProviderFailureKind.rateLimited,
          ),
        );
      } finally {
        dio.close(force: true);
      }
    },
    skip: Platform.environment['KONGROO_LIVE_MIRURO'] == '1'
        ? false
        : 'Set KONGROO_LIVE_MIRURO=1 to issue one ordinary public GET.',
  );
}
