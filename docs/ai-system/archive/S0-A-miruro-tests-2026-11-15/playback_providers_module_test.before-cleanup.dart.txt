import 'package:anime_app/data/player/providers/miruro/miruro_playback_provider.dart';
import 'package:anime_app/data/player/providers/playback_providers_module.dart';
import 'package:flutter_test/flutter_test.dart';

void main() {
  test('registers the Miruro adapter in the existing provider manager', () {
    final PlaybackProvidersModule module = PlaybackProvidersModule.open(
      clock: () => DateTime.utc(2026, 11, 15),
    );
    addTearDown(module.close);

    expect(module.manager.diagnostics, hasLength(1));
    expect(
      module.manager.diagnostics.single.id,
      MiruroPlaybackProvider.providerId,
    );
    expect(
      module.manager.provider(MiruroPlaybackProvider.providerId),
      isA<MiruroPlaybackProvider>(),
    );
  });

  test('closes its bounded HTTP client idempotently', () {
    final PlaybackProvidersModule module = PlaybackProvidersModule.open();

    expect(module.close, returnsNormally);
    expect(module.close, returnsNormally);
  });
}
